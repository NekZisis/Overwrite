import { Player } from './Player.js';
import {Background} from './Background.js';
import { InputHandler } from './InputHandler.js';

export class Game {
    constructor(width, height) {
        this.width = width;
        this.height = height;
        this.input = new InputHandler();
        this.player = new Player(this.width, this.height);
        this.background = new Background(this.width, this.height);
    }

    update(deltaTime) {
        this.player.update(deltaTime, this.input);
    }

    render(ctx) {
        // clear dynamic canvas
        ctx.clearRect(0, 0, this.width, this.height);

        // render entities (just the player for now)
        this.player.draw(ctx);
    }

    renderBackground(ctx) {
        ctx.clearRect(0, 0, this.width, this.height);

        // render all background 
        this.background.draw(ctx);
    }
}

export class InputHandler {
    constructor() {
        this.keys = new Set();

        // KEYBOARD

        window.addEventListener('keydown', (e) => {
        // Prevent browser viewport scrolling on navigation keys
            if (['Space', 'ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(e.code)) {
                e.preventDefault();
            }
            this.keys.add(e.code);
        });

        window.addEventListener('keyup', (e) => {
        this.keys.delete(e.code);
        });

    }

    isPressed(code) {
        return this.keys.has(code);
    }
}

import { Game } from './Game.js';

// static/background (bg) canvas
const bgCanvas = document.getElementById('bgCanvas');
const bgCTX = bgCanvas.getContext('2d');
bgCanvas.width = 1980;
bgCanvas.height = 1024;

// dynamic canvas
const gameCanvas = document.getElementById('gameCanvas');
const gameCTX = gameCanvas.getContext('2d');
gameCanvas.width = 1980;
gameCanvas.height = 1024;

// Initiate game
const game = new Game(gameCanvas.width, gameCanvas.height);
// render background once
game.renderBackground(bgCTX);

let lastTime = 0; // initialize the deltaTime calculator
function gameLoop(timeStamp) {
    const deltaTime = (timeStamp - lastTime) / 1000;
    lastTime = timeStamp;

    // game loop
    game.update(deltaTime); // update state
    game.render(gameCTX);  // render the new frame
    requestAnimationFrame(gameLoop); // continue
}

// block out all mobile users
function isMobileDevice() {
    const hasTouch = 'ontouchstart' in window || navigator.maxTouchPoints > 0;
    const isSmallScreen = window.innerWidth < 768;
    return hasTouch && isSmallScreen;
}
if (isMobileDevice()) {
    document.getElementById('mobile-blocker').style.display = 'flex';
} else {
    // starts game loop if not a mobile user
    requestAnimationFrame(gameLoop);
}

export class Player {
    constructor(gameWidth, gameHeight) {
        this.gameWidth = gameWidth;
        this.gameHeight = gameHeight;

        // player dimensions (hitbox)
        this.width = 30;
        this.height = 50;

        // spawn position (bottom-center for now)
        this.x = this.gameWidth / 2 - this.width / 2;
        this.y = this.gameHeight - this.height;

        // velocity vectors (px/s)
        this.vx = 0;
        this.vy = 0;

        // physics and control constants
        this.speed = 700;        // horizontal movement speed (px/s)
        this.jumpForce = -750;   // upward jump speed (px/s)
        this.downForce = 650;   //  downward movement speed (px/s)
        this.gravity = 1500;     // acceleration due to gravity (px/s^2)
        this.isOnGround = true;
        this.facing = "right";
    }

    update(deltaTime, input) {
        // 1. Horizontal Input Handling
        const leftPressed = input.isPressed('KeyA');
        const rightPressed = input.isPressed('KeyD');

        if (leftPressed && !rightPressed) {
          this.vx = -this.speed;
          this.facing = "left";
        } else if (rightPressed && !leftPressed) {
          this.vx = this.speed;
          this.facing = "right";
        } else {
          this.vx = 0;
        }

        // 2. Vertical / Jump Input Handling
        const jumpPressed = input.isPressed('KeyW');

        if (jumpPressed && this.isOnGround) {
          this.vy = this.jumpForce;
          this.isOnGround = false;
        }

        // 2.5 Vertical / Downwards Input Handling
        const downPressed = input.isPressed('KeyS');

        if (downPressed && !this.isOnGround) {
            this.vy = this.downForce;
        }

        // 3. Physics Integration
        this.vy += this.gravity * deltaTime;
        this.x += this.vx * deltaTime;
        this.y += this.vy * deltaTime;

        // 4. Horizontal Screen Boundaries
        if (this.x < 0) this.x = 0;
        if (this.x + this.width > this.gameWidth) {
          this.x = this.gameWidth - this.width;
        }

        // 5. Floor Collision & Ground Snap
        const floorLevel = this.gameHeight;

        if (this.y + this.height >= floorLevel) {
          this.y = floorLevel - this.height;
          this.vy = 0;
          this.isOnGround = true;
        }

        //6. Dash 
        const spacePressed = input.isPressed('Space');

        if (spacePressed) {
            // implement dash
        }
    }

    draw(ctx) {
        ctx.fillStyle = 'white';
        ctx.fillRect(this.x, this.y, this.width, this.height);
    }
}
